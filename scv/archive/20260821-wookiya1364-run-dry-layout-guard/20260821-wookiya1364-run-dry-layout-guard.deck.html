echo "OK [T1-T2]"
bash tests/run.sh >/dev/null || fail "T3 tests/run.sh"
for t in core/tests/test-*.sh; do bash "$t" >/dev/null || fail "T3 $t"; done
echo "ALL GATES OK"
'
